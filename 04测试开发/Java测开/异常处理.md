import java.util.Scanner;

// 1. 自定义异常：成绩超出范围
class ScoreOutOfRangeException extends Exception {
    public ScoreOutOfRangeException(String message) {
        super(message);
    }
}

public class ScoreManagement {
    // 2. 成绩录入方法（可能抛出异常）
    public static void inputScore(String name, int score) throws ScoreOutOfRangeException {
        // 检查姓名是否为空
        if (name == null || name.trim().isEmpty()) {
            throw new IllegalArgumentException("姓名不能为空！");
        }
        // 检查成绩是否合法
        if (score < 0 || score > 100) {
            throw new ScoreOutOfRangeException("成绩不合法：" + score + "（必须在0-100之间）");
        }
        // 合法时提示录入成功
        System.out.println("录入成功：" + name + "，成绩：" + score);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("请输入学生姓名：");
        String name = scanner.nextLine();

        System.out.print("请输入学生成绩：");
        int score = scanner.nextInt();

        // 3. 处理异常
        try {
            inputScore(name, score);
        } catch (ScoreOutOfRangeException e) {
            // 处理自定义异常
            System.out.println("错误提示：" + e.getMessage());
        } catch (IllegalArgumentException e) {
            // 处理系统异常
            System.out.println("错误提示：" + e.getMessage());
        } finally {
            // 4. 无论是否异常，都执行
            System.out.println("本次录入结束");
            scanner.close(); // 释放资源
        }
    }
}
